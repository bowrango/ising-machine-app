% Why the SYNC net carries 2f with no SYNC applied.
% The tank terminals A_L and A_R swing in antiphase; every inverter returns its current through
% GND pin 10, which is the SYNC net (R12||L12, ~50 ohm to ground). Each side's clamp conducts at
% the bottom of its swing, so the ground current gets one pulse from L and one from R per period:
% it repeats every T/2, the f term cancels, and 2f is the lowest harmonic (same for VCC / 3V3 ripple).
% Build: lualatex sync-ground-bounce.tex && pdftoppm -png -r 300 -singlefile sync-ground-bounce.pdf sync-ground-bounce
\documentclass[border=8pt]{standalone}
\usepackage[siunitx, american]{circuitikz}
\usepackage{fontspec}
\setmainfont{TeX Gyre Heros}
\usepackage{sansmath}
\sansmath
\sisetup{mode=text}

\definecolor{sideL}{HTML}{1F77B4}
\definecolor{sideR}{HTML}{FF7F0E}
\definecolor{loopc}{HTML}{2CA02C}

\begin{document}
\ctikzset{bipoles/thickness=1}
\begin{circuitikz}[
    font=\large,
    line width=1.2pt,
    box/.style={draw, line width=1.2pt, fill=white, minimum width=3.4cm, minimum height=1.5cm},
    wave/.style={line width=1.4pt, samples=120, smooth},
    guide/.style={draw=black!35, line width=0.6pt, densely dashed},
  ]

  % Oscillators -- SYNC net -- Filter -- source
  \node[box] (osc) at (2.5,2.6) {Oscillators};
  \node[box] (filt) at (9,2.6) {Filter};
  \draw (osc.east) -- (filt.west);
  \node[font=\normalsize\bfseries, above] at (4.9,2.6) {SYNC};
  \fill[loopc] (6.6,2.6) circle (0.12);
  \node[loopc, above, font=\normalsize\bfseries] at (6.6,2.65) {C3};
  \draw (filt.south) -- (9,0.7) node[ground, scale=1.6, line width=0.9pt]{};
  \node[font=\normalsize, anchor=west] at (9.15,1.3) {$v_\mathrm{SYNC} \approx \SI{50}{\ohm}\cdot i_\mathrm{GND}$};
  \draw (osc.south) -- (2.5,0.7) node[ground, scale=1.6, line width=0.9pt]{};
  % SYNC source, switched off
  \draw (filt.east) -- (11.3,2.6) to[nos, l=off] (13,2.6) -- (13,1.85);
  \draw (13,1.05) -- (13,0.7) node[ground, scale=1.6, line width=0.9pt]{};
  \draw[fill=white] (13,1.45) circle (0.4);
  \draw[-{Triangle[length=2.4mm, width=2.4mm]}, line width=2pt] (13,1.15) -- (13,1.75);

  % Circulating ground current
  \draw[loopc, line width=4pt, -{Triangle[length=3.5mm, width=5mm]}]
    (5.75,1.25) ++(100:0.6) arc[start angle=100, end angle=-40, radius=0.6];
  \draw[loopc, line width=4pt, -{Triangle[length=3.5mm, width=5mm]}]
    (5.75,1.25) ++(-80:0.6) arc[start angle=-80, end angle=-220, radius=0.6];

  % Waveforms: two periods, T = 2 cm, time origin at x0
  \def\xo{3.4}
  \begin{scope}[shift={(\xo,0)}]
    % A_L and A_R: antiphase
    \draw[wave, sideL] plot[domain=0:4] (\x, {6.4 + 0.4*sin(180*\x)});
    \draw[wave, sideR] plot[domain=0:4] (\x, {5.3 - 0.4*sin(180*\x)});
    \node[sideL, anchor=east] at (-0.1,6.4) {$A_L$ (C1)};
    \node[sideR, anchor=east] at (-0.1,5.3) {$A_R$ (C2)};
    \node[anchor=west] at (4.15,5.85) {$f$};

    % T/2 shift between the troughs
    \draw[{Latex[length=1.6mm]}-{Latex[length=1.6mm]}, line width=0.7pt] (1.5,6.95) -- node[above, font=\normalsize] {$T/2$} (2.5,6.95);
    \draw[guide] (1.5,6.0) -- (1.5,6.95);
    \draw[guide] (2.5,4.9) -- (2.5,6.95);

    % Ground current: one pulse at the bottom of each side's swing
    \foreach \t in {0.5, 2.5} \draw[guide] (\t,4.9) -- (\t,3.95);
    \foreach \t in {1.5, 3.5} \draw[guide] (\t,6.0) -- (\t,3.95);
    \draw[loopc, line width=1.4pt] (0,3.8) -- (4,3.8);
    \foreach \t/\c in {0.5/sideR, 1.5/sideL, 2.5/sideR, 3.5/sideL}
      \draw[wave, \c, samples=60] plot[domain={\t-0.35}:{\t+0.35}] (\x, {3.8 + 0.55*exp(-((\x-\t)/0.1)^2)});
    \node[loopc, anchor=east] at (-0.1,3.95) {$i_\mathrm{GND}$};
    \node[loopc, anchor=west, font=\large\bfseries] at (4.15,3.95) {$2f$};
  \end{scope}
\end{circuitikz}
\end{document}
