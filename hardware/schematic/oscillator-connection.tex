% Two oscillators i and j, each two SN74HC240 inverters cross-coupled through R = 1k onto an L || C tank
% (L = 30 uH, C = 750 pF; as in sim/oscillator.cir), with terminals L and R. The HC240 boxes show the inverters
% are one IC whose GND pin is the SYNC net, where the 2f0 SYNC is injected. Coupler R_ij joins i's R to j's L.
% Each oscillator's TLV3502 compares R (+IN) against L (-IN) and gives its phase bit OUT (WIN on the schematic, WIN_DETECT_X4.SchDoc).
% Build: lualatex oscillator-connection.tex && pdftoppm -png -r 300 -singlefile oscillator-connection.pdf oscillator-connection
\documentclass[border=8pt]{standalone}
\usepackage[american]{circuitikz}
\usepackage{fontspec}
\setmainfont{TeX Gyre Heros}
\usepackage{sansmath}
\sansmath

\begin{document}
\ctikzset{bipoles/thickness=1, tripoles/thickness=1}
\begin{circuitikz}[font=\large, line width=0.9pt, x=1.5cm, y=1.4cm]
  % One oscillator: terminal L at x=0, R at x=4; comparator below
  \newcommand{\osc}[2]{%
    \begin{scope}[shift={(#1,0)}]
      \node[above left] at (0,0) {L};
      \node[above right] at (4,0) {R};
      \draw (0,0) -- (0,-3.4) (4,0) -- (4,-3.4);
      % SN74HC240: both inverters in one package, its GND pin on the SYNC rail
      \node[not port, fill=white] (U1) at (2,0) {};
      \node[not port, xscale=-1, fill=white] (U2) at (2,-1.2) {};
      \draw[line width=1.6pt] (1.45,0.5) rectangle (2.55,-1.7);
      \draw (2,0.5) -- (2,1.2);
      \fill (2,1.2) circle (2pt);
      \draw (0,0) -- (U1.in);
      \draw (U1.out) -- (2.6,0) to[R, l=$R$] (4,0);
      \draw (0,-1.2) to[R, l=$R$] (1.4,-1.2) -- (U2.out);
      \draw (U2.in) -- (4,-1.2);
      \node[font=\Large] at (-0.55,-1.85) {$#2$};
      \draw (0,-2.3) to[L, l=$L$] (4,-2.3);
      \draw (0,-3.4) to[C, l=$C$] (4,-3.4);
      \foreach \y in {-1.2,-2.3,-3.4} { \fill (0,\y) circle (2pt); \fill (4,\y) circle (2pt); }
      % Comparator: + from R, - from L
      \node[op amp, noinv input up, rotate=-90] (W) at (2,-5.0) {};
      \draw (4,-3.4) -- (4,-4.0) -| (W.+);
      \draw (0,-3.4) -- (0,-4.0) -| (W.-);
      \draw (W.out) -- ++(0,-0.3) node[below] {OUT$_{#2}$};
    \end{scope}}

  \osc{0}{i}
  \osc{6}{j}

  % SYNC rail: both HC240 GND pins, driven at 2f0
  \draw (2,1.2) -- (11.2,1.2) to[sV] (11.2,-0.4) node[ground, scale=1.4, line width=0.7pt]{};
  \node[anchor=west, align=left] at (11.55,0.4) {SYNC\\$2f_0$};

  % Coupler: i.R -- R_ij -- j.L
  \draw[european resistors] (4,-4.0) to[vR, l=$R_{ij}$, *-*] (6,-4.0);

\end{circuitikz}
\end{document}
